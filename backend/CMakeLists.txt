find_package(OpenSSL REQUIRED COMPONENTS Crypto)
add_library(blueprint_core src/cards.cpp src/model.cpp src/config.cpp)
target_include_directories(blueprint_core PUBLIC include)
target_compile_features(blueprint_core PUBLIC cxx_std_20)
set_target_properties(blueprint_core PROPERTIES CXX_EXTENSIONS OFF)
target_link_libraries(blueprint_core PRIVATE OpenSSL::Crypto)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(blueprint_core PRIVATE -Wall -Wextra -Wpedantic -Wconversion)
endif()
add_executable(blueprint src/main.cpp)
target_link_libraries(blueprint PRIVATE blueprint_core)
if(BUILD_TESTING)
    find_package(Python3 REQUIRED COMPONENTS Interpreter)
    add_executable(blueprint_tests tests/test_core.cpp)
    target_link_libraries(blueprint_tests PRIVATE blueprint_core)
    add_test(NAME core COMMAND blueprint_tests)
    add_test(NAME model_import COMMAND Python3::Interpreter ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_models.py $<TARGET_FILE:blueprint> $<TARGET_FILE:blueprint_tests>)
endif()
set(BLUEPRINT_FULL_MODEL_CONFIG "" CACHE FILEPATH "Optional configuration for external complete-model validation")
if(BUILD_TESTING AND BLUEPRINT_FULL_MODEL_CONFIG)
    add_test(NAME full_models COMMAND blueprint verify-models --config ${BLUEPRINT_FULL_MODEL_CONFIG})
    set_tests_properties(full_models PROPERTIES LABELS integration)
endif()
